cmake_minimum_required(VERSION 3.20)
project(physicsnemo_qa_consumer LANGUAGES CXX)
include(GNUInstallDirs)

option(PNMIR_QA_WITH_CUDA "Test the installed AOTI and TensorRT SDK" ON)
if(PNMIR_QA_WITH_CUDA)
  find_package(PhysicsNeMoInference CONFIG REQUIRED COMPONENTS aoti tensorrt)
else()
  find_package(PhysicsNeMoInference CONFIG REQUIRED)
endif()
add_executable(physicsnemo-qa-consumer consumer.cpp)
target_compile_features(physicsnemo-qa-consumer PRIVATE cxx_std_20)
target_link_libraries(physicsnemo-qa-consumer PRIVATE PhysicsNeMoInference::runtime)
if(PNMIR_QA_WITH_CUDA)
  target_compile_definitions(physicsnemo-qa-consumer PRIVATE PNMIR_QA_WITH_CUDA=1)
  target_link_libraries(physicsnemo-qa-consumer PRIVATE PhysicsNeMoInference::aoti PhysicsNeMoInference::tensorrt)
endif()
if(APPLE)
  set(pnmir_qa_install_rpath "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
  set(pnmir_qa_install_rpath "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
set_target_properties(physicsnemo-qa-consumer PROPERTIES
  INSTALL_RPATH "${pnmir_qa_install_rpath}"
  INSTALL_RPATH_USE_LINK_PATH FALSE)
install(TARGETS physicsnemo-qa-consumer RUNTIME DESTINATION bin)
