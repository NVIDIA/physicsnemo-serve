cmake_minimum_required(VERSION 3.24)
project(PhysicsNeMoWorkflow VERSION 0.1.0 LANGUAGES CXX)
if(WIN32)
  set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()
include(CTest)
include(GNUInstallDirs)
option(PNMS_ENABLE_AOTI "Enable the installed AOTI SDK backend" OFF)
option(PNMS_ENABLE_TENSORRT "Enable the installed TensorRT SDK backend" OFF)
option(PNMS_ENABLE_ONNXRUNTIME "Enable the installed ONNX Runtime SDK backend" OFF)
option(PNMS_ENABLE_MOCK_BACKEND "Enable CPU-only SDK test backend" ${BUILD_TESTING})
set(PNMS_CUSTOM_WORKFLOW_SOURCE_DIR "" CACHE PATH "Source-linked custom workflow CMake project")
set(PNMS_CUSTOM_WORKFLOW_TARGET "" CACHE STRING "Custom workflow target defining create_custom_workflow")
set(components)
foreach(backend aoti tensorrt onnxruntime)
  string(TOUPPER "${backend}" upper)
  if(PNMS_ENABLE_${upper})
    list(APPEND components ${backend})
  endif()
endforeach()
find_package(PhysicsNeMoInference 0.1 CONFIG REQUIRED COMPONENTS ${components})
find_package(nlohmann_json 3.11 QUIET)
if(NOT nlohmann_json_FOUND)
  include(FetchContent)
  FetchContent_Declare(nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz
    URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(nlohmann_json)
endif()
add_library(physicsnemo_workflow_authoring INTERFACE)
add_library(PhysicsNeMoWorkflow::authoring ALIAS physicsnemo_workflow_authoring)
target_compile_features(physicsnemo_workflow_authoring INTERFACE cxx_std_20)
target_include_directories(physicsnemo_workflow_authoring INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(physicsnemo_workflow_authoring INTERFACE PhysicsNeMoInference::runtime)
set_target_properties(physicsnemo_workflow_authoring PROPERTIES EXPORT_NAME authoring)
add_library(physicsnemo_workflow SHARED src/host.cpp)
add_library(PhysicsNeMoWorkflow::host ALIAS physicsnemo_workflow)
target_link_libraries(physicsnemo_workflow PUBLIC PhysicsNeMoWorkflow::authoring
  PRIVATE nlohmann_json::nlohmann_json)
set_target_properties(physicsnemo_workflow PROPERTIES EXPORT_NAME host
  INSTALL_RPATH "$ORIGIN;$ORIGIN/../lib" INSTALL_RPATH_USE_LINK_PATH TRUE)
if(APPLE)
  set_target_properties(physicsnemo_workflow PROPERTIES
    INSTALL_NAME_DIR "@rpath" INSTALL_RPATH "@loader_path;@loader_path/../lib")
endif()
if(PNMS_ENABLE_MOCK_BACKEND)
  target_compile_definitions(physicsnemo_workflow PRIVATE PNMS_HAS_MOCK=1)
endif()
foreach(backend IN LISTS components)
  string(TOUPPER "${backend}" upper)
  target_compile_definitions(physicsnemo_workflow PRIVATE PNMS_HAS_${upper}=1)
  target_link_libraries(physicsnemo_workflow PRIVATE PhysicsNeMoInference::${backend})
endforeach()
if(PNMS_CUSTOM_WORKFLOW_SOURCE_DIR OR PNMS_CUSTOM_WORKFLOW_TARGET)
  if(NOT PNMS_CUSTOM_WORKFLOW_SOURCE_DIR OR NOT PNMS_CUSTOM_WORKFLOW_TARGET)
    message(FATAL_ERROR "Set both PNMS_CUSTOM_WORKFLOW_SOURCE_DIR and PNMS_CUSTOM_WORKFLOW_TARGET")
  endif()
  add_subdirectory("${PNMS_CUSTOM_WORKFLOW_SOURCE_DIR}" custom-workflow)
  if(NOT TARGET "${PNMS_CUSTOM_WORKFLOW_TARGET}")
    message(FATAL_ERROR "Custom workflow target does not exist: ${PNMS_CUSTOM_WORKFLOW_TARGET}")
  endif()
  target_link_libraries(physicsnemo_workflow PRIVATE "${PNMS_CUSTOM_WORKFLOW_TARGET}")
  target_compile_definitions(physicsnemo_workflow PRIVATE PNMS_HAS_CUSTOM_WORKFLOW=1)
endif()
add_executable(pnms_native_run src/main.cpp)
set_target_properties(pnms_native_run PROPERTIES OUTPUT_NAME pnms-native-run
  INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}" INSTALL_RPATH_USE_LINK_PATH TRUE)
if(APPLE)
  set_target_properties(pnms_native_run PROPERTIES
    INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
endif()
target_link_libraries(pnms_native_run PRIVATE PhysicsNeMoWorkflow::host)
install(TARGETS physicsnemo_workflow physicsnemo_workflow_authoring
  EXPORT PhysicsNeMoWorkflowTargets
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(TARGETS pnms_native_run RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT PhysicsNeMoWorkflowTargets NAMESPACE PhysicsNeMoWorkflow::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PhysicsNeMoWorkflow)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/PhysicsNeMoWorkflowConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/PhysicsNeMoWorkflowConfig.cmake"
  INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PhysicsNeMoWorkflow)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/PhysicsNeMoWorkflowConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/PhysicsNeMoWorkflowConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/PhysicsNeMoWorkflowConfigVersion.cmake"
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/PhysicsNeMoWorkflow)
if(BUILD_TESTING)
  if(NOT PNMS_ENABLE_MOCK_BACKEND)
    message(FATAL_ERROR "CPU workflow tests require PNMS_ENABLE_MOCK_BACKEND")
  endif()
  add_executable(workflow_host_tests tests/host_test.cpp)
  target_link_libraries(workflow_host_tests PRIVATE physicsnemo_workflow nlohmann_json::nlohmann_json)
  add_test(NAME workflow_host_tests COMMAND workflow_host_tests)
  if(PNMS_CUSTOM_WORKFLOW_TARGET STREQUAL "pnms_test_custom")
    add_executable(workflow_custom_tests tests/custom_test.cpp)
    target_link_libraries(workflow_custom_tests PRIVATE physicsnemo_workflow nlohmann_json::nlohmann_json)
    add_test(NAME workflow_custom_tests COMMAND workflow_custom_tests)
  endif()
endif()
